echo Hello World!

:: Run a PowerShell command (get the current date and time)
echo Running a PowerShell command to get the current date and time...
powershell -Command "Get-Date"

:: Run another PowerShell command (display a message)
echo Running a PowerShell command to display a message...
powershell -Command "Write-Output 'This message is from PowerShell!'"

:: Decode and display the encoded string
echo Decoding the encoded string...
set "encoded=SgBnAGkAZwB0AEcAWAB6AHMAdwBiAGgAbQB1AFMASABJAE8AQQA="
for /f "delims=" %%i in ('powershell -Command "[System.Text.Encoding]::Unicode.GetString([System.Convert]::FromBase64String('%encoded%'))"') do set "decoded=%%i"
echo Decoded string: %decoded%

pause
